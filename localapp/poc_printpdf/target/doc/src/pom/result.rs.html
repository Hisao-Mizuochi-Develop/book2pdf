1use std::{
2	error,
3	fmt::{self, Display},

6/// Parser error.
7#[derive(Debug, PartialEq, Clone)]
8pub enum Error {
9	Incomplete,
10	Mismatch {
11		message: String,
12		position: usize,
13	},
14	Conversion {
15		message: String,
16		position: usize,
17	},
18	Expect {
19		message: String,
20		position: usize,
21		inner: Box<Error>,
22	},
23	Custom {
24		message: String,
25		position: usize,
26		inner: Option<Box<Error>>,
27	},
28}
29
30impl error::Error for Error {
31	fn description(&self) -> &'static str {
32		"Parse error"
33	}
34}
35
36impl Display for Error {
37	fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
38		match self {
39			Self::Incomplete => write!(f, "Incomplete"),
40			Self::Mismatch {
41				ref message,
42				ref position,
43			} => write!(f, "Mismatch at {}: {}", position, message),
44			Self::Conversion {
45				ref message,
46				ref position,
47			} => write!(f, "Conversion failed at {}: {}", position, message),
48			Self::Expect {
49				ref message,
50				ref position,
51				ref inner,
52			} => write!(f, "{} at {}: {}", message, position, inner),
53			Self::Custom {
54				ref message,
55				ref position,
56				inner: Some(ref inner),
57			} => write!(f, "{} at {}, (inner: {})", message, position, inner),
58			Self::Custom {
59				ref message,
60				ref position,
61				inner: None,
62			} => write!(f, "{} at {}", message, position),
63		}
64	}
65}

67/// Parser result, `Result<O>` ia alias of `Result<O, pom::Error>`.
68pub type Result<O> = ::std::result::Result<O, Error>;
