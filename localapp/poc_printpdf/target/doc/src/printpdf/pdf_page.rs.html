13/// PDF page
14#[derive(Debug, Clone)]
15pub struct PdfPage {
16    /// The index of the page in the document
17    pub(crate) index: usize,
18    /// page width in point
19    pub width: Pt,
20    /// page height in point
21    pub height: Pt,
22    /// Page layers
23    pub layers: Vec<PdfLayer>,
24    /// Resources used in this page
25    pub(crate) resources: PdfResources,
26    /// Extend the page with custom ad-hoc attributes, as an escape hatch to the low level lopdf library.
27    /// Can be used to add annotations to a page.
28    /// If your dictionary is wrong it will produce a broken PDF without warning or useful messages.
29    pub(crate) extend_with: Option<lopdf::Dictionary>,
30}

32/// A "reference" to the current page, allows for inner mutability
33/// but only inside this library
34pub struct PdfPageReference {
35    /// A weak reference to the document, for inner mutability
36    pub document: Weak<RefCell<PdfDocument>>,
37    /// The index of the page this layer is on
38    pub page: PdfPageIndex,
39}

41impl PdfPage {
42    /// Create a new page, notice that width / height are in millimeter.
43    /// Page must contain at least one layer
44    #[inline]
45    pub fn new<S>(width: Mm, height: Mm, layer_name: S, page_index: usize) -> (Self, PdfLayerIndex)
46    where
47        S: Into<String>,
48    {
49        let mut page = Self {
50            index: page_index,
51            width: width.into(),
52            height: height.into(),
53            layers: Vec::new(),
54            resources: PdfResources::new(),
55            extend_with: None
56        };
57
58        let initial_layer = PdfLayer::new(layer_name);
59        page.layers.push(initial_layer);
60
61        let layer_index = page.layers.len() - 1;
62
63        (page, PdfLayerIndex(layer_index))
64    }
65
66    /// Iterates through the layers attached to this page and gathers all resources,
67    /// which the layers need. Then returns a dictonary with all the resources
68    /// (fonts, image XObjects, etc.)
69    ///
70    /// While originally I had planned to build a system where you can reference contents
71    /// from all over the document, this turned out to be a problem, because each type had
72    /// to be handled differently (PDF weirdness)
73    ///
74    /// `layers` should be a Vec with all layers (optional content groups) that were added
75    /// to the document on a document level, it should contain the indices of the layers
76    /// (they will be ignored, todo) and references to the actual OCG dictionaries
77    #[inline]
78    pub(crate) fn collect_resources_and_streams(
79        self,
80        doc: &mut lopdf::Document,
81        layers: &[(usize, lopdf::Object)],
82    ) -> (lopdf::Dictionary, Vec<lopdf::Stream>) {
83        let cur_layers = layers.iter().map(|l| l.1.clone()).collect();
84        let (resource_dictionary, ocg_refs) = self
85            .resources
86            .into_with_document_and_layers(doc, cur_layers);
87
88        // set contents
89        let mut layer_streams = Vec::<lopdf::Stream>::new();
90        use lopdf::content::Operation;
91        use lopdf::Object::*;
92
93        for (idx, mut layer) in self.layers.into_iter().enumerate() {
94            // push OCG and q to the beginning of the layer
95            layer.operations.insert(0, Operation::new("q", vec![]));
96            layer.operations.insert(
97                0,
98                Operation::new(
99                    "BDC",
100                    vec![Name("OC".into()), Name(ocg_refs[idx].name.clone().into())],
101                ),
102            );
103
104            // push OCG END and Q to the end of the layer stream
105            layer.operations.push(Operation::new("Q", vec![]));
106            layer.operations.push(Operation::new("EMC", vec![]));
107
108            // should end up looking like this:
109
110            // /OC /MC0 BDC
111            // q
112            // <layer stream content>
113            // Q
114            // EMC
115
116            let layer_stream = layer.into();
117            layer_streams.push(layer_stream);
118        }
119
120        (resource_dictionary, layer_streams)
121    }
122
123    /// Change the graphics state. Before this operation is done, you should save
124    /// the graphics state using the `save_graphics_state()` function. This will change the
125    /// current graphics state until the end of the page or until the page is reset to the
126    /// previous state.
127    /// Returns the old graphics state, in case it was overwritten, as well as a reference
128    /// to the currently active graphics state
129    #[inline]
130    pub fn add_graphics_state(
131        &mut self,
132        added_state: ExtendedGraphicsState,
133    ) -> ExtendedGraphicsStateRef {
134        self.resources.add_graphics_state(added_state)
135    }
136
137    /// __STUB__: Adds a pattern to the pages resources
138    #[inline]
139    pub fn add_pattern(&mut self, pattern: Pattern) -> PatternRef {
140        self.resources.add_pattern(pattern)
141    }
142
143    /// __STUB__: Adds an XObject to the pages resources.
144    /// __NOTE__: Watch out for scaling. Your XObject might be invisible or only 1pt x 1pt big
145    #[inline]
146    pub fn add_xobject(&mut self, xobj: XObject) -> XObjectRef {
147        self.resources.add_xobject(xobj)
148    }
149
150    /// __STUB__: Adds a Link Annotation to the pages resources.
151    #[inline]
152    pub fn add_link_annotation(&mut self, annotation: LinkAnnotation) -> LinkAnnotationRef {
153        self.resources.add_link_annotation(annotation)
154    }

159impl PdfPageReference {
160    /// Adds a page and returns the index of the currently added page
161    #[inline]
162    pub fn add_layer<S>(&self, layer_name: S) -> PdfLayerReference
163    where
164        S: Into<String>,
165    {
166        let doc = self.document.upgrade().unwrap();
167        let mut doc = doc.borrow_mut();
168        let page = &mut doc.pages[self.page.0];
169
170        let current_page_index = page.layers.len(); /* order is important */
171        let layer = PdfLayer::new(layer_name);
172        page.layers.push(layer);
173        let index = PdfLayerIndex(current_page_index);
174
175        PdfLayerReference {
176            document: self.document.clone(),
177            page: self.page,
178            layer: index,
179        }
180    }
181
182    /// Validates that a layer is present and returns a reference to it
183    #[inline]
184
185    pub fn get_layer(&self, layer: PdfLayerIndex) -> PdfLayerReference {
186        let doc = self.document.upgrade().unwrap();
187        let doc = doc.borrow();
188
189        let _ = &doc.pages[self.page.0].layers[layer.0];
190
191        PdfLayerReference {
192            document: self.document.clone(),
193            page: self.page,
194            layer,
195        }
196    }
197
198    #[inline]
199    pub fn extend_with(&self, dict: lopdf::Dictionary) {
200        let doc = self.document.upgrade().unwrap();
201        let mut doc = doc.borrow_mut();
202        let page = &mut doc.pages[self.page.0];
203        page.extend_with = Some(dict);
204    }
