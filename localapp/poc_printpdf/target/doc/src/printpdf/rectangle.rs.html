1//! Utilities for rectangle paths.
2
3use crate::{
4    path::{PaintMode, WindingOrder},
5    Mm, Point, OP_PATH_CONST_RECT, OP_PATH_PAINT_END, OP_PATH_PAINT_STROKE,
6};
7
8/// A helper struct to insert rectangular shapes into a PDF.
9///
10/// This can be used to paint rectangles or to clip other paths.
11#[derive(Debug, Copy, Clone)]
12pub struct Rect {
13    /// Position of the lower left point of the rectangle, relative to the bottom left corner of
14    /// the PDF page in pt.
15    pub ll: Point,
16    /// Position of the upper right point of the rectangle, relative to the bottom left corner of
17    /// the PDF page in pt.
18    pub ur: Point,
19    /// The paint mode of the rectangle.
20    pub mode: PaintMode,
21    /// The path-painting/clipping path operator.
22    pub winding: WindingOrder,
23}
24
25impl Rect {
26    /// Create a new point.
27    ///
28    /// **WARNING: The reference point for a point is the bottom left corner, not the top left**
29    #[inline]
30    pub fn new(llx: Mm, lly: Mm, urx: Mm, ury: Mm) -> Self {
31        Self {
32            ll: Point {
33                x: llx.into(),
34                y: lly.into(),
35            },
36            ur: Point {
37                x: urx.into(),
38                y: ury.into(),
39            },
40            mode: PaintMode::default(),
41            winding: WindingOrder::default(),
42        }
43    }
44
45    /// Returns a new `Rect` with the specified `mode`.
46    #[inline]
47    #[must_use]
48    pub fn with_mode(mut self, mode: PaintMode) -> Self {
49        self.mode = mode;
50        self
51    }
52
53    /// Returns a new `Rect` with the specified `winding`.
54    #[inline]
55    #[must_use]
56    pub fn with_winding(mut self, winding: WindingOrder) -> Self {
57        self.winding = winding;
58        self
59    }
60
61    /// Transform the `Rect` into a `Vec` of PDF [`Operation`]s.
62    ///
63    /// [`Operation`]: lopdf::content::Operation
64    #[must_use]
65    pub fn into_stream_op(self) -> Vec<lopdf::content::Operation> {
66        use lopdf::content::Operation;
67
68        let width = self.ur.x - self.ll.x;
69        let height = self.ur.y - self.ll.y;
70
71        let rect_op = Operation::new(
72            OP_PATH_CONST_RECT,
73            vec![
74                self.ll.x.into(),
75                self.ll.y.into(),
76                width.into(),
77                height.into(),
78            ],
79        );
80
81        let paint_op = match self.mode {
82            PaintMode::Clip => Operation::new(self.winding.get_clip_op(), vec![]),
83            PaintMode::Fill => Operation::new(self.winding.get_fill_op(), vec![]),
84            PaintMode::Stroke => Operation::new(OP_PATH_PAINT_STROKE, vec![]),
85            PaintMode::FillStroke => Operation::new(self.winding.get_fill_stroke_op(), vec![]),
86        };
87
88        if matches!(self.mode, PaintMode::Clip) {
89            vec![rect_op, paint_op, Operation::new(OP_PATH_PAINT_END, vec![])]
90        } else {
91            vec![rect_op, paint_op]
92        }
93    }
94}
95
96impl PartialEq for Rect {
97    // custom compare function because of floating point inaccuracy
98    fn eq(&self, other: &Rect) -> bool {
99        self.ll == other.ll && self.ur == other.ur
100    }
101}
