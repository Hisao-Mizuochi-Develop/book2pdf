1use crate::{Mm, Pt};
2
3#[derive(Debug, Copy, Clone)]
4pub struct Point {
5    /// x position from the bottom left corner in pt
6    pub x: Pt,
7    /// y position from the bottom left corner in pt
8    pub y: Pt,
9}
10
11impl Point {
12    /// Create a new point.
13    /// **WARNING: The reference point for a point is the bottom left corner, not the top left**
14    #[inline]
15    pub fn new(x: Mm, y: Mm) -> Self {
16        Self {
17            x: x.into(),
18            y: y.into(),
19        }
20    }
21}
22
23impl PartialEq for Point {
24    // custom compare function because of floating point inaccuracy
25    fn eq(&self, other: &Point) -> bool {
26        if self.x.0.is_normal()
27            && other.x.0.is_normal()
28            && self.y.0.is_normal()
29            && other.y.0.is_normal()
30        {
31            // four floating point numbers have to match
32            let x_eq = self.x == other.x;
33            if !x_eq {
34                return false;
35            }
36            let y_eq = self.y == other.y;
37            if y_eq {
38                return true;
39            }
40        }
41
42        false
43    }
44}
