3use crate::OffsetDateTime;

9/// Initial struct for Xmp metatdata. This should be expanded later for XML handling, etc.
10/// Right now it just fills out the necessary fields
11#[derive(Debug, Clone)]
12pub struct XmpMetadata {
13    /// Document ID
14    pub document_id: String,
15    /// Web-viewable or "default" or to be left empty. Usually "default".
16    pub rendition_class: Option<String>,
17    /// Document version
18    pub document_version: u32,
19}
20
21impl XmpMetadata {
22    /// Creates a new XmpMetadata object
23    pub fn new(rendition_class: Option<String>, document_version: u32) -> Self {
24        let document_id: String = random_character_string_32();
25        Self {
26            document_id,
27            rendition_class,
28            document_version,
29        }
30    }
31
32    /// Consumes the XmpMetadata and turns it into a PDF Object.
33    /// This is similar to the
34    pub(crate) fn into_obj(self, m: &PdfMetadata) -> lopdf::Object {
35        use lopdf::Object::*;
36        use lopdf::{Dictionary as LoDictionary, Stream as LoStream};
37
38        // Shared between XmpMetadata and DocumentInfo
39        let trapping = if m.trapping { "True" } else { "False" };
40
41        // let xmp_instance_id = "2898d852-f86f-4479-955b-804d81046b19";
42        let instance_id = random_character_string_32();
43        let create_date = to_pdf_xmp_date(&m.creation_date);
44        let modification_date = to_pdf_xmp_date(&m.modification_date);
45        let metadata_date = to_pdf_xmp_date(&m.metadata_date);
46
47        let pdf_x_version = m.conformance.get_identifier_string();
48        let document_version = self.document_version.to_string();
49        let document_id = self.document_id.to_string();
50
51        let rendition_class = match self.rendition_class {
52            Some(class) => class,
53            None => "".to_string(),
54        };
55
56        let xmp_metadata = format!(
57            include_str!("../assets/catalog_xmp_metadata.txt"),
58            create = create_date,
59            modify = modification_date,
60            mdate = metadata_date,
61            title = m.document_title,
62            id = document_id,
63            instance = instance_id,
64            class = rendition_class,
65            version = document_version,
66            pdfx = pdf_x_version,
67            trapping = trapping,
68            creator = m.creator,
69            subject = m.subject,
70            keywords = m.keywords.join(","),
71            identifier = m.identifier,
72            producer = m.producer
73        );
74
75        Stream(LoStream::new(
76            LoDictionary::from_iter(vec![("Type", "Metadata".into()), ("Subtype", "XML".into())]),
77            xmp_metadata.as_bytes().to_vec(),
78        ))
79    }
80}

82// D:2018-09-19T10:05:05+00'00'
83fn to_pdf_xmp_date(date: &OffsetDateTime) -> String {
84    // Since the time is in UTC, we know that the time zone
85    // difference to UTC is 0 min, 0 sec, hence the 00'00
86    format!(
87        "D:{:04}-{:02}-{:02}T{:02}:{:02}:{:02}+00'00'",
88        date.year(),
89        date.month(),
90        date.day(),
91        date.hour(),
92        date.minute(),
93        date.second(),
94    )
