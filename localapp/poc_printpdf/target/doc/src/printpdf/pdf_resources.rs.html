1use crate::{

8/// Struct for storing the PDF Resources, to be used on a PDF page
9#[derive(Default, Debug, Clone)]
10pub struct PdfResources {
11    /// External graphics objects
12    pub xobjects: XObjectList,
13    /// Patterns used on this page. Do not yet, use, placeholder.
14    pub patterns: PatternList,
15    /// Graphics states used on this page
16    pub graphics_states: ExtendedGraphicsStateList,
17    /// Layers / optional content ("Properties") in the resource dictionary
18    pub layers: OCGList,
19    /// Link Annotations used in this page
20    pub link_annotations: LinkAnnotationList,
21}
22
23impl PdfResources {
24    /// Creates a new PdfResources struct (resources for exactly one PDF page)
25    pub fn new() -> Self {
26        Self::default()
27    }
28
29    /// Add a graphics state to the resources
30    #[inline]
31    pub fn add_graphics_state(
32        &mut self,
33        added_state: ExtendedGraphicsState,
34    ) -> ExtendedGraphicsStateRef {
35        self.graphics_states.add_graphics_state(added_state)
36    }
37
38    /// Adds an XObject to the page
39    #[inline]
40    pub fn add_xobject(&mut self, xobj: XObject) -> XObjectRef {
41        self.xobjects.add_xobject(xobj)
42    }
43
44    /// __STUB__: Adds a pattern to the resources, to be used like a color
45    #[inline]
46    pub fn add_pattern(&mut self, pattern: Pattern) -> PatternRef {
47        self.patterns.add_pattern(pattern)
48    }
49
50    /// __STUB__: Adds a link annotation to the resources
51    #[inline]
52    pub fn add_link_annotation(&mut self, link_annotation: LinkAnnotation) -> LinkAnnotationRef {
53        self.link_annotations.add_link_annotation(link_annotation)
54    }
55
56    /// See `XObject::Into_with_document`.
57    /// The resources also need access to the layers (the optional content groups), this should be a
58    /// `Vec<lopdf::Object::Reference>` (to the actual OCG groups, which are added on the document level)
59
60    pub fn into_with_document_and_layers(
61        self,
62        doc: &mut lopdf::Document,
63        layers: Vec<lopdf::Object>,
64    ) -> (lopdf::Dictionary, Vec<OCGRef>) {
65        let mut dict = lopdf::Dictionary::new();
66
67        let mut ocg_dict = self.layers;
68        let mut ocg_references = Vec::<OCGRef>::new();
69
70        let xobjects_dict: lopdf::Dictionary = self.xobjects.into_with_document(doc);
71        let patterns_dict: lopdf::Dictionary = self.patterns.into();
72        let graphics_state_dict: lopdf::Dictionary = self.graphics_states.into();
73        let annotations_dict: lopdf::Dictionary = self.link_annotations.into();
74
75        if !layers.is_empty() {
76            for l in layers {
77                ocg_references.push(ocg_dict.add_ocg(l));
78            }
79
80            let cur_ocg_dict_obj: lopdf::Dictionary = ocg_dict.into();
81
82            if !cur_ocg_dict_obj.is_empty() {
83                dict.set("Properties", lopdf::Object::Dictionary(cur_ocg_dict_obj));
84            }
85        }
86
87        if !xobjects_dict.is_empty() {
88            dict.set("XObject", lopdf::Object::Dictionary(xobjects_dict));
89        }
90
91        if !patterns_dict.is_empty() {
92            dict.set("Pattern", lopdf::Object::Dictionary(patterns_dict));
93        }
94
95        if !graphics_state_dict.is_empty() {
96            dict.set("ExtGState", lopdf::Object::Dictionary(graphics_state_dict));
97        }
98
99        if !annotations_dict.is_empty() {
100            dict.set("Annots", lopdf::Object::Dictionary(annotations_dict))
101        }
102
103        (dict, ocg_references)
104    }
105}
