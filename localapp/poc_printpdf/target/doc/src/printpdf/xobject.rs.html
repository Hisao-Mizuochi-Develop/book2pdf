16/// External object that gets reference outside the PDF content stream
17/// Gets constructed similar to the `ExtGState`, then inserted into the `/XObject` dictionary
18/// on the page. You can instantiate `XObjects` with the `/Do` operator. The `layer.add_xobject()`
19/// (or better yet, the `layer.add_image()`, `layer.add_form()`) methods will do this for you.
20#[derive(Debug, Clone)]
21pub enum XObject {
22    /* /Subtype /Image */
23    /// Image XObject, for images
24    Image(ImageXObject),
25    /* /Subtype /Form */
26    /// Form XObject, for PDF forms
27    Form(Box<FormXObject>),
28    /* /Subtype /PS */
29    /// Embedded PostScript XObject, for legacy applications
30    /// You can embed PostScript in a PDF, it is not recommended
31    PostScript(PostScriptXObject),
32    /// XObject embedded from an external stream
33    ///
34    /// This is mainly used to add XObjects to the resources that the library
35    /// doesn't support natively (such as gradients, patterns, etc).
36    ///
37    /// The only thing this does is to ensure that this stream is set on
38    /// the /Resources dictionary of the page. The `XObjectRef` returned
39    /// by `add_xobject()` is the unique name that can be used to invoke
40    /// the `/Do` operator (by the `use_xobject`)
41    External(LoPdfStream),
42}
43
44impl From<ImageXObject> for XObject {
45    fn from(i: ImageXObject) -> XObject {
46        XObject::Image(i)
47    }
48}

54        stream
55    }

60        let _ = stream.compress();
61        stream

65impl From<XObject> for lopdf::Object {
66    fn from(val: XObject) -> Self {
67        match val {
68            XObject::Image(image) => lopdf::Object::Stream(XObject::compress_stream(image.into())),
69            XObject::Form(form) => {
70                let cur_form: FormXObject = *form;
71                lopdf::Object::Stream(XObject::compress_stream(cur_form.into()))
72            }
73            XObject::PostScript(ps) => lopdf::Object::Stream(XObject::compress_stream(ps.into())),
74            XObject::External(stream) => lopdf::Object::Stream(XObject::compress_stream(stream)),
75        }
76    }
77}

83}

85impl XObjectList {
86    /// Creates a new XObjectList
87    pub fn new() -> Self {
88        Self::default()
89    }
90
91    /// Adds a new XObject to the list
92    pub fn add_xobject(&mut self, xobj: XObject) -> XObjectRef {
93        let len = self.objects.len();
94        let xobj_ref = XObjectRef::new(len);
95        self.objects.insert(xobj_ref.name.clone(), xobj);
96        xobj_ref
97    }
98
99    /// Same as `Into<lopdf::Dictionary>`, but since the dictionary
100    /// items in an XObject dictionary are streams and must be added to
101    /// the document as __references__, this function needs an additional
102    /// access to the PDF document so that we can add the streams first and
103    /// then track the references to them.
104
105    pub fn into_with_document(self, doc: &mut lopdf::Document) -> lopdf::Dictionary {
106        self.objects
107            .into_iter()
108            .map(|(name, object)| {
109                let obj: lopdf::Object = object.into();
110                let obj_ref = doc.add_object(obj);
111                (name, lopdf::Object::Reference(obj_ref))
112            })
113            .collect()
114    }
115}

196                LimitErrorKind::InsufficientMemory,

198        }
199
200        let mut image_data = vec![0; num_image_bytes as usize];
201        image.read_image(&mut image_data)?;
202
203        let (processed_color_type, processed_image_data, smask) = preprocess_image_with_alpha(
204            color_type,
205            image_data,
206            dim
207        );
208
209        let color_bits = ColorBits::from(processed_color_type);
210        let color_space = ColorSpace::from(processed_color_type);
211
212        Ok(Self {
213            width: Px(dim.0 as usize),
214            height: Px(dim.1 as usize),
215            color_space,
216            bits_per_component: color_bits,
217            image_data: processed_image_data,

256impl From<ImageXObject> for lopdf::Stream {
257    fn from(img: ImageXObject) -> lopdf::Stream {
258        use lopdf::Object::*;
259
260        let cs: &'static str = img.color_space.into();
261        let bbox: lopdf::Object = img.clipping_bbox.unwrap_or(CurTransMat::Identity).into();
262
263        let smask = if let Some(mask) = img.smask {
264            // This is using the "Soft-Mask Images" approach. See page 447 of the adobe PDF 1.4 reference
265            XObject::Image(ImageXObject {
266                width: img.width,
267                height: img.width,
268                color_space: ColorSpace::Greyscale,
269                bits_per_component: ColorBits::Bit8,
270                interpolate: false,
271                image_data: mask.matte.into_iter().map(|i| i as u8).collect(),
272                image_filter: None,
273                smask: None,
274                clipping_bbox: None,
275            })
276            .into()
277        } else {
278            Null
279        };
280
281        let mut dict = lopdf::Dictionary::from_iter(vec![
282            ("Type", Name("XObject".as_bytes().to_vec())),
283            ("Subtype", Name("Image".as_bytes().to_vec())),
284            ("Width", Integer(img.width.0 as i64)),
285            ("Height", Integer(img.height.0 as i64)),
286            ("Interpolate", img.interpolate.into()),
287            ("BitsPerComponent", Integer(img.bits_per_component.into())),
288            ("ColorSpace", Name(cs.as_bytes().to_vec())),
289            ("SMask", smask),
290            ("BBox", bbox),
291        ]);
292
293        if let Some(filter) = img.image_filter {
294            let params = match filter {
295                // TODO technically we could use multiple filters,
296                // DCT as an exception!
297                ImageFilter::DCT => {
298                    vec![
299                        ("Filter", Array(vec![Name("DCTDecode".as_bytes().to_vec())])),
300                        // not necessary, unless missing in the jpeg header
301                        (
302                            "DecodeParams",
303                            Dictionary(lopdf::dictionary!("ColorTransform" => Integer(0))),
304                        ),
305                    ]
306                }
307                _ => unimplemented!("Encountered filter type is not supported"),
308            };
309
310            params
311                .into_iter()
312                .for_each(|param| dict.set(param.0, param.1));
313        }
314
315        lopdf::Stream::new(dict, img.image_data)
316    }
317}

339/// __THIS IS NOT A PDF FORM!__ A form `XObject` can be nearly everything.
340/// PDF allows you to reuse content for the graphics stream in a `FormXObject`.
341/// A `FormXObject` is basically a layer-like content stream and can contain anything
342/// as long as it's a valid strem. A `FormXObject` is intended to be used for reapeated
343/// content on one page.
344#[derive(Debug, Clone)]
345pub struct FormXObject {
346    /* /Type /XObject */
347    /* /Subtype /Form */
348
349    /* /FormType Integer */
350    /// Form type (currently only Type1)
351    pub form_type: FormType,
352    /// The actual content of this FormXObject
353    pub bytes: Vec<u8>,
354    /* /Matrix [Integer , 6] */
355    /// Optional matrix, maps the form into user space
356    pub matrix: Option<CurTransMat>,
357    /* /Resources << dictionary >> */
358    /// (Optional but strongly recommended; PDF 1.2) A dictionary specifying
359    /// any resources (such as fonts and images) required by the form XObject
360    /// (see Section 3.7, “Content Streams and Resources”).
361    ///
362    /// In PDF 1.1 and earlier, all named resources used in the form XObject must be
363    /// included in the resource dictionary of each page object on which the form
364    /// XObject appears, regardless of whether they also appear in the resource
365    /// dictionary of the form XObject. It can be useful to specify these resources
366    /// in the form XObject’s resource dictionary as well, to determine which resources
367    /// are used inside the form XObject. If a resource is included in both dictionaries,
368    /// it should have the same name in both locations.
369    ///  /// In PDF 1.2 and later versions, form XObjects can be independent of the content
370    /// streams in which they appear, and this is strongly recommended although not
371    /// required. In an independent form XObject, the resource dictionary of the form
372    /// XObject is required and contains all named resources used by the form XObject.
373    /// These resources are not promoted to the outer content stream’s resource
374    /// dictionary, although that stream’s resource dictionary refers to the form XObject.
375    pub resources: Option<lopdf::Dictionary>,
376    /* /Group << dictionary >> */
377    /// (Optional; PDF 1.4) A group attributes dictionary indicating that the contents of the
378    /// form XObject are to be treated as a group and specifying the attributes of that group
379    /// (see Section 4.9.2, “Group XObjects”).
380    ///
381    /// Note: If a Ref entry (see below) is present, the group attributes also apply to the
382    /// external page imported by that entry, which allows such an imported page to be treated
383    /// as a group without further modification.
384    pub group: Option<GroupXObject>,
385    /* /Ref << dictionary >> */
386    /// (Optional; PDF 1.4) A reference dictionary identifying a page to be imported from another
387    /// PDF file, and for which the form XObject serves as a proxy (see Section 4.9.3, “Reference XObjects”).
388    pub ref_dict: Option<lopdf::Dictionary>,
389    /* /Metadata [stream] */
390    /// (Optional; PDF 1.4) A metadata stream containing metadata for the form XObject
391    /// (see Section 10.2.2, “Metadata Streams”).
392    pub metadata: Option<lopdf::Stream>,
393    /* /PieceInfo << dictionary >> */
394    /// (Optional; PDF 1.3) A page-piece dictionary associated with the form XObject
395    /// (see Section 10.4, “Page-Piece Dictionaries”).
396    pub piece_info: Option<lopdf::Dictionary>,
397    /* /LastModified (date) */
398    /// (Required if PieceInfo is present; optional otherwise; PDF 1.3) The date and time
399    /// (see Section 3.8.3, “Dates”) when the form XObject’s contents were most recently
400    /// modified. If a page-piece dictionary (PieceInfo) is present, the modification date
401    /// is used to ascertain which of the application data dictionaries it contains correspond
402    /// to the current content of the form (see Section 10.4, “Page-Piece Dictionaries”).
403    pub last_modified: Option<OffsetDateTime>,
404    /* /StructParent integer */
405    /// (Required if the form XObject is a structural content item; PDF 1.3) The integer key of
406    /// the form XObject’s entry in the structural parent tree (see “Finding Structure Elements
407    /// from Content Items” on page 868).
408    pub struct_parent: Option<i64>,
409    /* /StructParents integer */
410    /// __(Required if the form XObject contains marked-content sequences that are structural content
411    /// items; PDF 1.3)__ The integer key of the form XObject’s entry in the structural parent tree
412    /// (see “Finding Structure Elements from Content Items” on page 868).
413    ///
414    /// __Note:__ At most one of the entries StructParent or StructParents may be present. A form
415    /// XObject can be either a content item in its entirety or a container for marked-content sequences
416    /// that are content items, but not both.
417    pub struct_parents: Option<i64>,
418    /* /OPI << dictionary >> */
419    /// (Optional; PDF 1.2) An OPI version dictionary for the form XObject
420    /// (see Section 10.10.6, “Open Prepress Interface (OPI)”).
421    pub opi: Option<lopdf::Dictionary>,
422    /// (Optional; PDF 1.5) An optional content group or optional content membership dictionary
423    /// (see Section 4.10, “Optional Content”) specifying the optional content properties for
424    /// the form XObject. Before the form is processed, its visibility is determined based on
425    /// this entry. If it is determined to be invisible, the entire form is skipped, as if there
426    /// were no Do operator to invoke it.
427    pub oc: Option<lopdf::Dictionary>,
428    /* /Name /MyName */
429    /// __(Required in PDF 1.0; optional otherwise)__ The name by which this form XObject is referenced
430    /// in the XObject subdictionary of the current resource dictionary
431    /// (see Section 3.7.2, “Resource Dictionaries”).
432    /// __Note:__ This entry is obsolescent and its use is no longer recommended.
433    /// (See implementation note 55 in Appendix H.)
434    pub name: Option<String>,
435}

637fn preprocess_image_with_alpha(color_type: ColorType, image_data: Vec<u8>, dim: (u32, u32)) -> (ColorType, Vec<u8>, Option<SMask>) {
638    match color_type {
639        image_crate::ColorType::Rgba8 => {
640            let (rgb, alpha) = rgba_to_rgb(image_data);
641            let smask = Some(SMask {
642                bits_per_component: 8,
643                width: dim.0 as i64,
644                height: dim.1 as i64,
645                interpolate: false,
646                matte: alpha.into_iter().map(|b| b as i64).collect(),
647            });
648            (ColorType::Rgb8, rgb, smask)
649        },
650        _ => (color_type, image_data, None)
651    }
652}
