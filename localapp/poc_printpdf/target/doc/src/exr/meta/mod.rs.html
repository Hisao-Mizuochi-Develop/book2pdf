26/// Contains the complete meta data of an exr image.
27///
28/// Defines how the image is split up in the file,
29/// the number and type of images and channels,
30/// and various other attributes.
31/// The usage of custom attributes is encouraged.
32#[derive(Debug, Clone, PartialEq)]
33pub struct MetaData {
34    /// Some flags summarizing the features that must be supported to decode the
35    /// file.
36    pub requirements: Requirements,
37
38    /// One header to describe each layer in this file.
39    // TODO rename to layer descriptions?
40    pub headers: Headers,
41}
42
43/// List of `Header`s.
44pub type Headers = SmallVec<[Header; 3]>;
45
46/// List of `OffsetTable`s.
47pub type OffsetTables = SmallVec<[OffsetTable; 3]>;
48
49/// The offset table is an ordered list of indices referencing pixel data in the
50/// exr file.
51///
52/// For each pixel tile in the image, an index exists, which points to the
53/// byte-location of the corresponding pixel data in the file. That index can be
54/// used to load specific portions of an image without processing all bytes in a
55/// file. For each header, an offset table exists with its indices ordered by
56/// `LineOrder::Increasing`.
57// If the multipart bit is unset and the chunkCount attribute is not present,
58// the number of entries in the chunk table is computed using the
59// dataWindow, tileDesc, and compression attribute.
60//
61// If the multipart bit is set, the header must contain a
62// chunkCount attribute, that contains the length of the offset table.
63pub type OffsetTable = Vec<u64>;

65/// A summary of requirements that must be met to read this exr file.
66///
67/// Used to determine whether this file can be read by a given reader.
68/// It includes the `OpenEXR` version number. This library aims to support
69/// version `2.0`.
70#[derive(Clone, Copy, Eq, PartialEq, Debug, Hash)]
71pub struct Requirements {
72    /// This library supports reading version 1 and 2, and writing version 2.
73    // TODO write version 1 for simple images
74    pub file_format_version: u8,
75
76    /// If true, this image has tiled blocks and contains only a single layer.
77    /// If false and not deep and not multilayer, this image is a single layer
78    /// image with scan line blocks.
79    pub is_single_layer_and_tiled: bool,
80
81    // in c or bad c++ this might have been relevant (omg is he allowed to say that)
82    /// Whether this file has strings with a length greater than 31.
83    /// Strings can never be longer than 255.
84    pub has_long_names: bool,
85
86    /// This image contains at least one layer with deep data.
87    pub has_deep_data: bool,
88
89    /// Whether this file contains multiple layers.
90    pub has_multiple_layers: bool,
91}

153/// The first four bytes of each exr file.
154/// Used to abort reading non-exr files.
155pub mod magic_number {
156    use super::*;
157
158    /// The first four bytes of each exr file.
159    pub const BYTES: [u8; 4] = [0x76, 0x2f, 0x31, 0x01];
160
161    /// Without validation, write this instance to the byte stream.
162    pub fn write(write: &mut impl Write) -> Result<()> {
163        u8::write_slice_ne(write, &self::BYTES)
164    }
165
166    /// Consumes four bytes from the reader and returns whether the file may be
167    /// an exr file.
168    // TODO check if exr before allocating BufRead
169    pub fn is_exr(read: &mut impl Read) -> Result<bool> {
170        let mut magic_num = [0; 4];
171        u8::read_slice_ne(read, &mut magic_num)?;
172        Ok(magic_num == self::BYTES)
173    }
174
175    /// Validate this image. If it is an exr file, return `Ok(())`.
176    pub fn validate_exr(read: &mut impl Read) -> UnitResult {
177        if self::is_exr(read)? {
178            Ok(())
179        } else {
180            Err(Error::invalid("file identifier missing"))
181        }
182    }

209/// Compute the number of tiles required to contain all values.
210pub fn compute_block_count(full_res: usize, tile_size: usize) -> usize {
211    // round up, because if the image is not evenly divisible by the tiles,
212    // we add another tile at the end (which is only partially used)
213    RoundingMode::Up.divide(full_res, tile_size)
214}

216/// Compute the start position and size of a block inside a dimension.
217#[inline]
218pub fn calculate_block_position_and_size(
219    total_size: usize,
220    block_size: usize,
221    block_index: usize,
222) -> Result<(usize, usize)> {
223    let block_position = block_size * block_index;
224
225    Ok((block_position, calculate_block_size(total_size, block_size, block_position)?))
226}

228/// Calculate the size of a single block. If this is the last block,
229/// this only returns the required size, which is always smaller than the
230/// default block size.
231// TODO use this method everywhere instead of convoluted formulas
232#[inline]
233pub fn calculate_block_size(
234    total_size: usize,
235    block_size: usize,
236    block_position: usize,
237) -> Result<usize> {
238    if block_position >= total_size {
239        return Err(Error::invalid(format!(
240            "block position {block_position} exceeds total size {total_size}"
241        )));
242    }
243
244    if block_position + block_size <= total_size {
245        Ok(block_size)
246    } else {
247        Ok(total_size - block_position)
248    }
249}

251/// Calculate number of mip levels in a given resolution.
252// TODO this should be cached? log2 may be very expensive
253pub fn compute_level_count(round: RoundingMode, full_res: usize) -> usize {
254    usize::try_from(round.log2(u32::try_from(full_res).unwrap())).unwrap() + 1
255}

257/// Calculate the size of a single mip level by index.
258// TODO this should be cached? log2 may be very expensive
259pub fn compute_level_size(round: RoundingMode, full_res: usize, level_index: usize) -> usize {
260    assert!(
261        level_index < std::mem::size_of::<usize>() * 8,
262        "largest level size exceeds maximum integer value"
263    );
264    round.divide(full_res, 1 << level_index).max(1)
265}

301/// Iterates over all rip map level indices of a given size.
302/// The order of iteration conforms to `LineOrder::Increasing`.
303pub fn rip_map_indices(
304    round: RoundingMode,
305    max_resolution: Vec2<usize>,
306) -> impl Iterator<Item = Vec2<usize>> {
307    let (width, height) = (
308        compute_level_count(round, max_resolution.width()),
309        compute_level_count(round, max_resolution.height()),
310    );
311
312    (0..height).flat_map(move |y_level| (0..width).map(move |x_level| Vec2(x_level, y_level)))
313}

315/// Iterates over all mip map level indices of a given size.
316/// The order of iteration conforms to `LineOrder::Increasing`.
317pub fn mip_map_indices(
318    round: RoundingMode,
319    max_resolution: Vec2<usize>,
320) -> impl Iterator<Item = usize> {
321    0..compute_level_count(round, max_resolution.width().max(max_resolution.height()))
322}

324/// Compute the number of chunks that an image is divided into. May be an
325/// expensive operation.
326// If not multilayer and chunkCount not present,
327// the number of entries in the chunk table is computed
328// using the dataWindow and tileDesc attributes and the compression format
329pub fn compute_chunk_count(
330    compression: Compression,
331    data_size: Vec2<usize>,
332    blocks: BlockDescription,
333) -> usize {
334    if let BlockDescription::Tiles(tiles) = blocks {
335        let round = tiles.rounding_mode;
336        let Vec2(tile_width, tile_height) = tiles.tile_size;
337
338        // TODO cache all these level values??
339        use crate::meta::attribute::LevelMode::*;
340        match tiles.level_mode {
341            Singular => {
342                let tiles_x = compute_block_count(data_size.width(), tile_width);
343                let tiles_y = compute_block_count(data_size.height(), tile_height);
344                tiles_x * tiles_y
345            }
346
347            MipMap => mip_map_levels(round, data_size)
348                .map(|(_, Vec2(level_width, level_height))| {
349                    compute_block_count(level_width, tile_width)
350                        * compute_block_count(level_height, tile_height)
351                })
352                .sum(),
353
354            RipMap => rip_map_levels(round, data_size)
355                .map(|(_, Vec2(level_width, level_height))| {
356                    compute_block_count(level_width, tile_width)
357                        * compute_block_count(level_height, tile_height)
358                })
359                .sum(),
360        }
361    }
362    // scan line blocks never have mip maps
363    else {
364        compute_block_count(data_size.height(), compression.scan_lines_per_block())
365    }
366}

402        magic_number::validate_exr(read)?;

406        // do this check now in order to fast-fail for newer versions and features than
407        // version 2

409
410        let headers = Header::read_all(read, &requirements, pedantic)?;
411
412        // TODO check if supporting requirements 2 always implies supporting
413        // requirements 1
414        Ok(Self {
415            requirements,
416            headers,
417        })

517    /// Validates this meta data. Returns the minimal possible requirements.
518    pub fn validate(headers: &[Header], pedantic: bool) -> Result<Requirements> {
519        if headers.is_empty() {
520            return Err(Error::invalid("at least one layer is required"));
521        }
522
523        let deep = false; // TODO deep data
524        let is_multilayer = headers.len() > 1;
525        let first_header_has_tiles =
526            headers.iter().next().map_or(false, |header| header.blocks.has_tiles());
527
528        let mut minimal_requirements = Requirements {
529            // according to the spec, version 2  should only be necessary if `is_multilayer ||
530            // deep`. but the current open exr library does not support images with
531            // version 1, so always use version 2.
532            file_format_version: 2,
533
534            // start as low as possible, later increasing if required
535            has_long_names: false,
536
537            is_single_layer_and_tiled: !is_multilayer && first_header_has_tiles,
538            has_multiple_layers: is_multilayer,
539            has_deep_data: deep,
540        };
541
542        for header in headers {
543            if header.deep {
544                // TODO deep data (and then remove this check)
545                return Err(Error::unsupported("deep data not supported yet"));
546            }
547
548            header.validate(is_multilayer, &mut minimal_requirements.has_long_names, pedantic)?;
549        }
550
551        // TODO validation fn!
552        // if let Some(max) = max_pixel_bytes {
553        // let byte_size: usize = headers.iter()
554        // .map(|header| header.total_pixel_bytes())
555        // .sum();
556        //
557        // if byte_size > max {
558        // return Err(Error::invalid("image larger than specified maximum"));
559        // }
560        // }
561
562        if pedantic {
563            // check for duplicate header names
564            let mut header_names = HashSet::with_capacity(headers.len());
565            for header in headers {
566                if !header_names.insert(&header.own_attributes.layer_name) {
567                    return Err(Error::invalid(format!(
568                        "duplicate layer name: `{}`",
569                        header.own_attributes.layer_name.as_ref().expect("header validation bug")
570                    )));
571                }
572            }
573        }
574
575        if pedantic {
576            let must_share =
577                headers.iter().flat_map(|header| header.own_attributes.other.iter()).any(
578                    |(_, value)| value.to_chromaticities().is_ok() || value.to_time_code().is_ok(),
579                );
580
581            if must_share {
582                return Err(Error::invalid("chromaticities and time code attributes must must not exist in own attributes but shared instead"));
583            }
584        }
585
586        if pedantic && headers.len() > 1 {
587            // check for attributes that should not differ in between headers
588            let first_header = headers.first().expect("header count validation bug");
589            let first_header_attributes = &first_header.shared_attributes;
590
591            for header in &headers[1..] {
592                if &header.shared_attributes != first_header_attributes {
593                    return Err(Error::invalid("display window, pixel aspect, chromaticities, and time code attributes must be equal for all headers"));
594                }
595            }
596        }
597
598        debug_assert!(minimal_requirements.validate().is_ok(), "inferred requirements are invalid");
599        Ok(minimal_requirements)
600    }
601}

607    pub fn is_multilayer(&self) -> bool {
608        self.has_multiple_layers

611    /// Read the value without validating.
612    pub fn read<R: Read>(read: &mut R) -> Result<Self> {
613        use ::bit_field::BitField;
614
615        let version_and_flags = u32::read_le(read)?;
616
617        // take the 8 least significant bits, they contain the file format version
618        // number
619        let version = (version_and_flags & 0x000F) as u8;
620
621        // the 24 most significant bits are treated as a set of boolean flags
622        let is_single_tile = version_and_flags.get_bit(9);
623        let has_long_names = version_and_flags.get_bit(10);
624        let has_deep_data = version_and_flags.get_bit(11);
625        let has_multiple_layers = version_and_flags.get_bit(12);
626
627        // all remaining bits except 9, 10, 11 and 12 are reserved and should be 0
628        // if a file has any of these bits set to 1, it means this file contains
629        // a feature that we don't support
630        let unknown_flags = version_and_flags >> 13; // all flags excluding the 12 bits we already parsed
631
632        if unknown_flags != 0 {
633            // TODO test if this correctly detects unsupported files
634            return Err(Error::unsupported("too new file feature flags"));
635        }
636
637        let version = Self {
638            file_format_version: version,
639            is_single_layer_and_tiled: is_single_tile,
640            has_long_names,
641            has_deep_data,
642            has_multiple_layers,
643        };
644
645        Ok(version)
646    }

648    /// Without validation, write this instance to the byte stream.
649    pub fn write<W: Write>(self, write: &mut W) -> UnitResult {
650        use ::bit_field::BitField;
651
652        // the 8 least significant bits contain the file format version number
653        // and the flags are set to 0
654        let mut version_and_flags = u32::from(self.file_format_version);
655
656        // the 24 most significant bits are treated as a set of boolean flags
657        version_and_flags.set_bit(9, self.is_single_layer_and_tiled);
658        version_and_flags.set_bit(10, self.has_long_names);
659        version_and_flags.set_bit(11, self.has_deep_data);
660        version_and_flags.set_bit(12, self.has_multiple_layers);
661        // all remaining bits except 9, 10, 11 and 12 are reserved and should be 0
662
663        version_and_flags.write_le(write)?;
664        Ok(())
665    }
666
667    /// Validate this instance.
668    pub fn validate(&self) -> UnitResult {
669        if self.file_format_version == 2 {
670            match (
671                self.is_single_layer_and_tiled,
672                self.has_deep_data,
673                self.has_multiple_layers,
674                self.file_format_version,
675            ) {
676                // Single-part scan line. One normal scan line image.
677                (false, false, false, 1..=2) => Ok(()),
678
679                // Single-part tile. One normal tiled image.
680                (true, false, false, 1..=2) => Ok(()),
681
682                // Multi-part (new in 2.0).
683                // Multiple normal images (scan line and/or tiled).
684                (false, false, true, 2) => Ok(()),
685
686                // Single-part deep data (new in 2.0).
687                // One deep tile or deep scan line part
688                (false, true, false, 2) => Ok(()),
689
690                // Multi-part deep data (new in 2.0).
691                // Multiple parts (any combination of:
692                // tiles, scan lines, deep tiles and/or deep scan lines).
693                (false, true, true, 2) => Ok(()),
694
695                _ => Err(Error::invalid("file feature flags")),
696            }
697        } else {
698            Err(Error::unsupported("file versions other than 2.0 are not supported"))
699        }
700    }
701}
