22/// Shared by this compression method and DWA's RLE scheme (both port OpenEXR's
23/// `internal_rle_decompress`, see `compression::dwa`) - kept separate from
24/// `decompress_bytes` because DWA does not apply the delta prediction /
25/// byte-block interleaving done there.
26pub(super) fn unpack_rle_tokens(
27    compressed_le: &[u8],
28    expected_byte_size: usize,
29    pedantic: bool,
30) -> Result<ByteVec> {
31    let mut remaining_le = compressed_le;
32    let mut decompressed_le = Vec::with_capacity(expected_byte_size.min(8 * 2048));
33
34    while !remaining_le.is_empty() && decompressed_le.len() != expected_byte_size {
35        let count = take_1(&mut remaining_le)? as i8 as i32;
36
37        if count < 0 {
38            // take the next '-count' bytes as-is
39            let values = take_n(&mut remaining_le, -count as usize)?;
40            decompressed_le.extend_from_slice(values);
41        } else {
42            // repeat the next value 'count + 1' times
43            let value = take_1(&mut remaining_le)?;
44            decompressed_le.resize(decompressed_le.len() + (count as usize) + 1, value);
45        }
46    }
47
48    if pedantic && !remaining_le.is_empty() {
49        return Err(Error::invalid("data amount"));
50    }
51
52    Ok(decompressed_le)
53}

70/// Shared by this compression method and DWA's RLE section. This only emits
71/// the byte-oriented RLE token stream; callers are responsible for any byte
72/// prediction, byte interleaving, or zlib wrapping required by their format.
73pub(super) fn pack_rle_tokens(data_le: &[u8]) -> ByteVec {
74    let mut compressed_le = Vec::with_capacity(data_le.len());
75    let mut run_start = 0;
76    let mut run_end = 1;
77
78    while run_start < data_le.len() {
79        while run_end < data_le.len()
80            && data_le[run_start] == data_le[run_end]
81            && ((run_end - run_start) as i32) - 1 < (MAX_RUN_LENGTH as i32)
82        {
83            run_end += 1;
84        }
85
86        if run_end - run_start >= MIN_RUN_LENGTH {
87            compressed_le.push((((run_end - run_start) as i32) - 1) as u8);
88            compressed_le.push(data_le[run_start]);
89            run_start = run_end;
90        } else {
91            while run_end < data_le.len()
92                && (run_end + 1 >= data_le.len()
93                    || data_le[run_end] != data_le[run_end + 1]
94                    || run_end + 2 >= data_le.len()
95                    || data_le[run_end + 1] != data_le[run_end + 2])
96                && run_end - run_start < MAX_RUN_LENGTH
97            {
98                run_end += 1;
99            }
100
101            compressed_le.push(((run_start as i32) - (run_end as i32)) as u8);
102            compressed_le.extend_from_slice(&data_le[run_start..run_end]);
103
104            run_start = run_end;
105            run_end += 1;
106        }
107    }
108
109    compressed_le
110}

112fn take_1(slice: &mut &[u8]) -> Result<u8> {
113    if !slice.is_empty() {
114        let result = slice[0];
115        *slice = &slice[1..];
116        Ok(result)
117    } else {
118        Err(Error::invalid("compressed data"))
119    }
120}
121
122fn take_n<'s>(slice: &mut &'s [u8], n: usize) -> Result<&'s [u8]> {
123    if n <= slice.len() {
124        let (front, back) = slice.split_at(n);
125        *slice = back;
126        Ok(front)
127    } else {
128        Err(Error::invalid("compressed data"))
129    }
