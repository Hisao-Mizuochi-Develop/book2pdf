23/// Y'CbCr -> R'G'B' inverse conversion for DWA, using the modified 709
24/// coefficients OpenEXR's DWA encoder uses. Input comp0/1/2 are Y, RY, BY;
25/// output is R, G, B.
26#[inline]
27fn csc709_inverse(comp0: f32, comp1: f32, comp2: f32) -> (f32, f32, f32) {
28    let r = comp0 + 1.5747 * comp2;
29    let g = comp0 - 0.1873 * comp1 - 0.4682 * comp2;
30    let b = comp0 + 1.8556 * comp1;
31    (r, g, b)
32}
33
34/// R'G'B' -> Y'CbCr forward conversion for DWA. The component order matches
35/// OpenEXR's channel-group storage: Y, BY, RY.
36#[inline]
37fn csc709_forward(r: f32, g: f32, b: f32) -> (f32, f32, f32) {
38    // OpenEXR uses a modified 709 transform with a zero-centered chroma
39    // representation instead of the usual 0.5 offset.
40    let y = 0.2126 * r + 0.7152 * g + 0.0722 * b;
41    let by = (b - y) / 1.8556;
42    let ry = (r - y) / 1.5747;
43    (y, by, ry)
44}

46pub(super) fn encode_lossy_channels(
47    infos: &[ChannelInfo],
48    csc_groups: &[[usize; 3]],
49    channel_bytes: &[Vec<u8>],
50    quant_base_error: f32,
51) -> Result<(Vec<u16>, Vec<u16>)> {
52    // Lossy chunks use shared AC/DC streams. CSC triplets consume the streams
53    // first, then standalone LOSSY_DCT channels continue from the same cursors.
54    let mut ac = Vec::new();
55    let mut dc = Vec::new();
56    let mut grouped = vec![false; infos.len()];
57
58    for &group in csc_groups {
59        let info = &infos[group[0]];
60        let components = group
61            .iter()
62            .map(|&channel| channel_half_samples(&channel_bytes[channel], &infos[channel], true))
63            .collect::<Result<Vec<_>>>()?;
64
65        encode_lossy_dct_group(
66            &components,
67            info.width,
68            info.height,
69            quant_base_error,
70            &mut ac,
71            &mut dc,
72        )?;
73
74        for &channel in &group {
75            grouped[channel] = true;
76        }
77    }
78
79    for (index, info) in infos.iter().enumerate() {
80        if grouped[index] || info.scheme != CompressorScheme::LossyDct {
81            continue;
82        }
83
84        let apply_nonlinear = !info.quantize_linearly;
85        let samples = channel_half_samples(&channel_bytes[index], info, apply_nonlinear)?;
86        encode_lossy_dct_group(
87            &[samples],
88            info.width,
89            info.height,
90            quant_base_error,
91            &mut ac,
92            &mut dc,
93        )?;
94    }
95
96    Ok((ac, dc))
97}

99fn channel_half_samples(
100    bytes: &[u8],
101    info: &ChannelInfo,
102    apply_nonlinear: bool,
103) -> Result<Vec<u16>> {
104    // OpenEXR stores lossy input as half precision internally before DCT.
105    // F32 channels are clamped to the finite half range and demoted here.
106    let mut samples = Vec::with_capacity(info.width * info.height);
107
108    match info.sample_type {
109        SampleType::F16 => {
110            let chunks = bytes.chunks_exact(2);
111            if !chunks.remainder().is_empty() {
112                return Err(Error::invalid("DWA f16 channel data size"));
113            }
114            samples.extend(chunks.map(|pair| u16::from_le_bytes([pair[0], pair[1]])));
115        }
116        SampleType::F32 => {
117            let chunks = bytes.chunks_exact(4);
118            if !chunks.remainder().is_empty() {
119                return Err(Error::invalid("DWA f32 channel data size"));
120            }
121            samples.extend(chunks.map(|quad| {
122                let mut value = f32::from_le_bytes([quad[0], quad[1], quad[2], quad[3]]);
123                value = value.clamp(-65504.0, 65504.0);
124                f16::from_f32(value).to_bits()
125            }));
126        }
127        SampleType::U32 => {
128            return Err(Error::unsupported("DWA lossy DCT compression of u32 channels"));
129        }
130    }
131
132    if samples.len() != info.width * info.height {
133        return Err(Error::invalid("DWA lossy channel data size mismatch"));
134    }
135
136    if apply_nonlinear {
137        let to_nonlinear = to_nonlinear_table();
138        for sample in &mut samples {
139            *sample = to_nonlinear[*sample as usize];
140        }
141    }
142
143    Ok(samples)
144}

146fn encode_lossy_dct_group(
147    components: &[Vec<u16>],
148    width: usize,
149    height: usize,
150    quant_base_error: f32,
151    ac: &mut Vec<u16>,
152    dc: &mut Vec<u16>,
153) -> Result<()> {
154    if width == 0 || height == 0 {
155        return Ok(());
156    }
157
158    let component_count = components.len();
159    if component_count != 1 && component_count != 3 {
160        return Err(Error::invalid("invalid DWA lossy component count"));
161    }
162
163    for component in components {
164        if component.len() != width * height {
165            return Err(Error::invalid("DWA lossy component size mismatch"));
166        }
167    }
168
169    // Mirror the source block edges so partial 8x8 blocks behave the same way
170    // as the reference encoder.
171    let quant_tables = QuantTables::new(quant_base_error);
172    let blocks_x = (width + 7) / 8;
173    let blocks_y = (height + 7) / 8;
174    let block_count = blocks_x * blocks_y;
175    let mut group_dc: Vec<Vec<u16>> =
176        (0..component_count).map(|_| Vec::with_capacity(block_count)).collect();
177
178    let mut row_blocks: Vec<[[f32; 64]; 3]> = vec![[[0.0; 64]; 3]; blocks_x];
179
180    for block_y in 0..blocks_y {
181        for block_x in 0..blocks_x {
182            for component_index in 0..component_count {
183                let block = &mut row_blocks[block_x][component_index];
184                for y in 0..8 {
185                    let src_y = mirror_index(block_y * 8 + y, height);
186                    for x in 0..8 {
187                        let src_x = mirror_index(block_x * 8 + x, width);
188                        let bits = components[component_index][src_y * width + src_x];
189                        block[y * 8 + x] = f16::from_bits(bits).to_f32();
190                    }
191                }
192            }
193
194            if component_count == 3 {
195                // CSC is performed in nonlinear space for the RGB triplet.
196                let dct_blocks = &mut row_blocks[block_x];
197                for i in 0..64 {
198                    let (y, by, ry) =
199                        csc709_forward(dct_blocks[0][i], dct_blocks[1][i], dct_blocks[2][i]);
200                    dct_blocks[0][i] = y;
201                    dct_blocks[1][i] = by;
202                    dct_blocks[2][i] = ry;
203                }
204            }
205        }
206
207        discrete_cosine_transform::dct_forward_8x8_batch(
208            row_blocks.iter_mut().flat_map(|blocks| blocks[..component_count].iter_mut()),
209        );
210
211        for block_x in 0..blocks_x {
212            for component_index in 0..component_count {
213                let block = &mut row_blocks[block_x][component_index];
214                let (tolerances, half_tolerances) = if component_index == 0 {
215                    (&quant_tables.y, &quant_tables.half_y)
216                } else {
217                    (&quant_tables.cbcr, &quant_tables.half_cbcr)
218                };
219
220                let half_zig = quantize_coefficients_to_zigzag(block, tolerances, half_tolerances);
221                group_dc[component_index].push(half_zig[0]);
222                rle_ac(&half_zig, ac);
223            }
224        }
225    }
226
227    for component_dc in group_dc {
228        dc.extend(component_dc);
229    }
230
231    Ok(())
232}

234fn mirror_index(index: usize, length: usize) -> usize {
235    // The C encoder mirrors out-of-bounds coordinates back into the image
236    // rather than clamping them. This keeps edge blocks symmetrical.
237    debug_assert_ne!(length, 0);
238    let mut value = index as isize;
239    let length = length as isize;
240
241    if value >= length {
242        value = length - (value - (length - 1));
243    }
244    if value < 0 {
245        value = length - 1;
246    }
247
248    value as usize
249}

251/// One of the chunk-global u16 streams (AC or DC). All channel groups of a
252/// chunk consume the same stream, so the cursor carries across groups.
253pub(super) struct PackedStream<'v> {
254    values: &'v [u16],
255    cursor: usize,
256}
257
258impl<'v> PackedStream<'v> {
259    fn new(values: &'v [u16]) -> Self {
260        Self {
261            values,
262            cursor: 0,
263        }
264    }
265
266    fn next(&mut self) -> Option<u16> {
267        let value = self.values.get(self.cursor).copied();
268        self.cursor += 1;
269        value
270    }
271
272    /// Value at "offset" past the cursor, without consuming (the DC stream
273    /// is indexed planar per group and advanced once at group end).
274    fn peek_at(&self, offset: usize) -> Option<u16> {
275        self.values.get(self.cursor + offset).copied()
276    }
277
278    fn advance(&mut self, count: usize) {
279        self.cursor += count;
280    }
281}

283/// Decode all LOSSY_DCT channels: first every CSC group, then the
284/// standalone channels, both in channel order - the order in which the
285/// encoder appended them to the shared AC/DC streams.
286pub(super) fn decode_lossy_channels(
287    infos: &[ChannelInfo],
288    csc_groups: &[[usize; 3]],
289    ac_packed: &[u16],
290    dc_packed: &[u16],
291) -> Result<Vec<Vec<f16>>> {
292    // Decode CSC triplets first, then standalone lossy channels. The shared
293    // AC/DC cursors advance in the same order the encoder wrote them.
294    let mut ac = PackedStream::new(ac_packed);
295    let mut dc = PackedStream::new(dc_packed);
296
297    let mut samples: Vec<Vec<f16>> = vec![vec![]; infos.len()];
298    let mut grouped = vec![false; infos.len()];
299
300    for &group in csc_groups {
301        // all three channels have identical sampling, hence identical size
302        let info = &infos[group[0]];
303        let mut decoded: [Vec<f16>; 3] =
304            std::array::from_fn(|_| vec![f16::ZERO; info.width * info.height]);
305
306        decode_lossy_dct_group(
307            &mut ac,
308            &mut dc,
309            info.width,
310            info.height,
311            Some(to_linear_table()),
312            &mut decoded,
313        )?;
314
315        for (&channel, channel_samples) in group.iter().zip(decoded) {
316            samples[channel] = channel_samples;
317            grouped[channel] = true;
318        }
319    }
320
321    for (index, info) in infos.iter().enumerate() {
322        if grouped[index] || info.scheme != CompressorScheme::LossyDct {
323            continue;
324        }
325        let mut decoded = [vec![f16::ZERO; info.width * info.height]];
326        let to_linear = (!info.quantize_linearly).then(to_linear_table);
327        decode_lossy_dct_group(&mut ac, &mut dc, info.width, info.height, to_linear, &mut decoded)?;
328
329        let [channel_samples] = decoded;
330        samples[index] = channel_samples;
331    }
332
333    Ok(samples)
334}

336/// Decode one standalone channel (decoded.len() == 1) or one CSC'd R/G/B
337/// triplet (decoded.len() == 3): per 8x8 block and component, read the
338/// DC value, un-RLE the AC values, inverse-DCT
339fn decode_lossy_dct_group(
340    ac: &mut PackedStream<'_>,
341    dc: &mut PackedStream<'_>,
342    width: usize,
343    height: usize,
344    to_linear: Option<&[u16; 65536]>,
345    decoded: &mut [Vec<f16>],
346) -> Result<()> {
347    let components = decoded.len();
348    let blocks_x = (width + 7) / 8;
349    let blocks_y = (height + 7) / 8;
350    let block_count = blocks_x * blocks_y;
351
352    // Buffer the whole group rather than one block at a time. That lets the
353    // inverse DCT batch over every block that actually needs it, matching the
354    // structure of the C reference.
355    let mut dct_blocks: Vec<[[f32; 64]; 3]> = vec![[[0.0f32; 64]; 3]; block_count];
356    let mut needs_inverse_dct: Vec<[bool; 3]> = vec![[false; 3]; block_count];
357
358    for block_y in 0..blocks_y {
359        for block_x in 0..blocks_x {
360            let block_index = block_y * blocks_x + block_x;
361
362            for component in 0..components {
363                let mut zig_block = [0u16; 64];
364
365                // the DC stream is planar: all of component 0's blocks,
366                // then all of component 1's, ...
367                zig_block[0] = dc
368                    .peek_at(component * block_count + block_index)
369                    .ok_or_else(|| Error::invalid("truncated DWA DC data"))?;
370
371                let last_non_zero = un_rle_ac(ac, &mut zig_block)?;
372
373                let dct_block = &mut dct_blocks[block_index][component];
374                if last_non_zero == 0 {
375                    // DC-only block: all AC coefficients are zero, so the
376                    // inverse DCT can fill the whole block from one value.
377                    dct_block[0] = f16::from_bits(zig_block[0]).to_f32();
378                    discrete_cosine_transform::dct_inverse_8x8_dc_only(dct_block);
379                } else {
380                    from_half_zigzag(&zig_block, dct_block);
381                    needs_inverse_dct[block_index][component] = true;
382                }
383            }
384        }
385    }
386
387    discrete_cosine_transform::dct_inverse_8x8_batch(
388        dct_blocks
389            .iter_mut()
390            .zip(needs_inverse_dct.iter())
391            .flat_map(|(blocks, flags)| blocks.iter_mut().zip(flags.iter()))
392            .filter_map(|(block, &needed)| needed.then_some(block)),
393    );
394
395    for block_y in 0..blocks_y {
396        for block_x in 0..blocks_x {
397            let block_index = block_y * blocks_x + block_x;
398            let dct_blocks = &mut dct_blocks[block_index];
399
400            if components == 3 {
401                for i in 0..64 {
402                    let (r, g, b) =
403                        csc709_inverse(dct_blocks[0][i], dct_blocks[1][i], dct_blocks[2][i]);
404                    dct_blocks[0][i] = r;
405                    dct_blocks[1][i] = g;
406                    dct_blocks[2][i] = b;
407                }
408            }
409
410            // Convert nonlinear DCT output back to linear half values and crop
411            // the edges to the actual image extent.
412            for (component, output) in decoded.iter_mut().enumerate() {
413                for y in block_y * 8..(block_y * 8 + 8).min(height) {
414                    for x in block_x * 8..(block_x * 8 + 8).min(width) {
415                        let value =
416                            dct_blocks[component][(y - block_y * 8) * 8 + (x - block_x * 8)];
417                        let nonlinear = f16::from_f32(value);
418                        output[y * width + x] = f16::from_bits(match to_linear {
419                            Some(to_linear) => to_linear[nonlinear.to_bits() as usize],
420                            None => nonlinear.to_bits(),
421                        });
422                    }
423                }
424            }
425        }
426    }
427
428    dc.advance(components * block_count);
429    Ok(())
430}

433mod test {
434    use rand::{Rng, SeedableRng};
435
436    use super::*;
437    use crate::image::validate_results::ValidateResult;

439    const SEED: [u8; 32] = [
440        66, 100, 19, 240, 8, 91, 3, 128, 9, 44, 201, 17, 88, 6, 255, 61, 30, 11, 2, 121, 99, 1,
441        250, 77, 33, 7, 42, 13, 200, 176, 22, 5,
442    ];
443
444    /// The R'G'B' <-> Y'CbCr conversion pair must round-trip: converting to
445    /// Y'CbCr and back must recover the original RGB triple (approximately,
446    /// since the matrix coefficients are not exactly invertible in f32). The
447    /// forward output tuple `(y, by, ry)` feeds the inverse positionally.
448    fn assert_csc_roundtrips(r: f32, g: f32, b: f32) {
449        let (y, by, ry) = csc709_forward(r, g, b);
450        let (r2, g2, b2) = csc709_inverse(y, by, ry);
451        vec![r, g, b].assert_approx_equals_result(&vec![r2, g2, b2]);
452    }
453
454    #[test]
455    fn csc_roundtrip_hardcoded() {
456        assert_csc_roundtrips(0.0, 0.0, 0.0);
457        assert_csc_roundtrips(1.0, 1.0, 1.0);
458        assert_csc_roundtrips(1.0, 0.0, 0.0);
459        assert_csc_roundtrips(0.0, 1.0, 0.0);
460        assert_csc_roundtrips(0.0, 0.0, 1.0);
461        assert_csc_roundtrips(0.25, 0.5, 0.75);
462    }
463
464    #[test]
465    fn csc_roundtrip_seeded() {
466        let mut random = rand::rngs::StdRng::from_seed(SEED);
467        for _ in 0..256 {
468            let mut channel = || random.gen_range(-4.0f32..4.0);
469            assert_csc_roundtrips(channel(), channel(), channel());
470        }
471    }
