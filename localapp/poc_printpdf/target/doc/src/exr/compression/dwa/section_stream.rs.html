1// The four on-disk DWA payload sections (UNKNOWN, AC, DC, RLE): splitting the
2// chunk body into them, the zlib inflate wrapper they share, and the
3// differencing transform applied to the DC stream.
4
5use super::chunk_header::{AcCompression, DwaHeader};
6use crate::error::{Error, Result};
7
8/// Split the data after header + rules into the four sections, in on-disk
9/// order. Errors on truncation like the C parser.
10pub(super) fn split_sections<'d>(data: &'d [u8], header: &DwaHeader) -> Result<[&'d [u8]; 4]> {
11    let mut rest = data;
12    let mut take = |length: usize| -> Result<&'d [u8]> {
13        if length > rest.len() {
14            return Err(Error::invalid("truncated DWA section"));
15        }
16        let (section, remaining) = rest.split_at(length);
17        rest = remaining;
18        Ok(section)
19    };
20
21    Ok([
22        take(header.unknown_compressed_size)?,
23        take(header.ac_compressed_size)?,
24        take(header.dc_compressed_size)?,
25        take(header.rle_compressed_size)?,
26    ])
27}
28
29fn inflate(compressed: &[u8], expected_size: usize) -> Result<Vec<u8>> {
30    let options = zune_inflate::DeflateOptions::default()
31        .set_limit(expected_size)
32        .set_size_hint(expected_size);
33
34    let inflated = zune_inflate::DeflateDecoder::new_with_options(compressed, options)
35        .decode_zlib()
36        .map_err(|_| Error::invalid("DWA zlib data malformed"))?;
37
38    if inflated.len() != expected_size {
39        return Err(Error::invalid("DWA zlib data size mismatch"));
40    }
41    Ok(inflated)
42}

44/// UNKNOWN section: raw (non-DCT-compressible) channel data,
45/// zlib-compressed, planar in channel order.
46pub(super) fn decode_unknown_section(section: &[u8], header: &DwaHeader) -> Result<Vec<u8>> {
47    if header.unknown_uncompressed_size == 0 {
48        return Ok(vec![]);
49    }
50    inflate(section, header.unknown_uncompressed_size)
51}
52
53/// AC section: RLE DCT coefficients as u16, entropy coded with either the
54/// PIZ static Huffman coder or zlib.
55pub(super) fn decode_ac_section(section: &[u8], header: &DwaHeader) -> Result<Vec<u16>> {
56    if header.ac_count == 0 {
57        return Ok(vec![]);
58    }
59
60    match header.ac_compression {
61        AcCompression::StaticHuffman => {
62            crate::compression::piz::huffman::decompress(section, header.ac_count)
63        }
64        AcCompression::Deflate => {
65            let bytes = inflate(section, header.ac_count * 2)?;
66            Ok(bytes.chunks_exact(2).map(|pair| u16::from_le_bytes([pair[0], pair[1]])).collect())
67        }
68    }
69}

71/// DC section: one u16 (half bits) per 8x8 block, zlib-compressed after
72/// the "zip reconstruct" transform (differencing + byte deinterleave).
73pub(super) fn decode_dc_section(section: &[u8], header: &DwaHeader) -> Result<Vec<u16>> {
74    if header.dc_count == 0 {
75        return Ok(vec![]);
76    }
77
78    let bytes = inflate(section, header.dc_count * 2)?;
79    let bytes = undo_zip_reconstruct(&bytes);
80    Ok(bytes.chunks_exact(2).map(|pair| u16::from_le_bytes([pair[0], pair[1]])).collect())
81}
82
83/// RLE section: zlib, then classic byte-oriented RLE. Result is planar per
84/// channel, each channel further split into byte planes.
85pub(super) fn decode_rle_section(section: &[u8], header: &DwaHeader) -> Result<Vec<u8>> {
86    if header.rle_raw_size == 0 {
87        return Ok(vec![]);
88    }
89    let inflated = inflate(section, header.rle_uncompressed_size)?;
90    crate::compression::rle::unpack_rle_tokens(&inflated, header.rle_raw_size, false)
91}

93/// Ports "internal_zip_reconstruct_bytes": undo differencing, then
94/// interleave the two buffer halves.
95fn undo_zip_reconstruct(source: &[u8]) -> Vec<u8> {
96    if source.len() < 2 {
97        return source.to_vec();
98    }
99
100    let mut deltas = source.to_vec();
101    for index in 1..deltas.len() {
102        deltas[index] = ((deltas[index - 1] as i32) + (deltas[index] as i32) - 128) as u8;
103    }
104
105    let (first_half, second_half) = deltas.split_at((deltas.len() + 1) / 2);
106    let mut out = vec![0u8; deltas.len()];
107    for (index, slot) in out.iter_mut().enumerate() {
108        *slot = if index % 2 == 0 {
109            first_half[index / 2]
110        } else {
111            second_half[index / 2]
112        };
113    }
114    out
115}

117/// Encoder-side companion applied to the DC byte stream before zlib:
118/// byte-fragment separation followed by successive differencing.
119pub(super) fn zip_deconstruct_bytes(bytes: &mut [u8]) {
120    crate::compression::optimize_bytes::separate_bytes_fragments(bytes);
121    crate::compression::optimize_bytes::samples_to_differences(bytes);
122}

124#[cfg(test)]
125mod test {
126    use rand::{Rng, SeedableRng};
127
128    use super::*;
129
130    const SEED: [u8; 32] = [
131        19, 240, 8, 91, 3, 128, 9, 44, 201, 17, 88, 6, 255, 61, 30, 11, 2, 121, 99, 1, 250, 77, 33,
132        7, 42, 13, 200, 176, 22, 5, 66, 100,
133    ];
134
135    /// The encoder-side DC transform (`zip_deconstruct_bytes`) and the
136    /// decoder-side reconstruction (`undo_zip_reconstruct`) must be inverses,
137    /// including for odd lengths and the < 2 byte short-circuit.
138    #[test]
139    fn zip_deconstruct_reconstruct_roundtrip() {
140        let mut random = rand::rngs::StdRng::from_seed(SEED);
141
142        for length in [0usize, 1, 2, 3, 4, 5, 17, 64, 129] {
143            let original: Vec<u8> = (0..length).map(|_| random.gen()).collect();
144
145            let mut deconstructed = original.clone();
146            zip_deconstruct_bytes(&mut deconstructed);
147            let reconstructed = undo_zip_reconstruct(&deconstructed);
148
149            assert_eq!(reconstructed, original, "failed at length {length}");
150        }
151    }
