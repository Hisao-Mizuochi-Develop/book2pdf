20/// Specify whether to read the image in parallel,
21/// whether to use pedantic error handling,
22/// and a callback for the reading progress.
23#[derive(Debug, Clone)]
24pub struct ReadImage<OnProgress, ReadLayers> {
25    on_progress: OnProgress,
26    read_layers: ReadLayers,
27    pedantic: bool,
28    parallel: bool,
29}
30
31impl<F, L> ReadImage<F, L>
32where
33    F: FnMut(f64),
34{
35    /// Uses relaxed error handling and parallel decompression.
36    pub fn new(read_layers: L, on_progress: F) -> Self {
37        Self {
38            on_progress,
39            read_layers,
40            pedantic: false,
41            #[cfg(not(feature = "rayon"))]
42            parallel: false,
43            #[cfg(feature = "rayon")]
44            parallel: true,
45        }
46    }
47
48    /// Specify that any missing or unusual information should result in an
49    /// error. Otherwise, `exrs` will try to compute or ignore missing
50    /// information.
51    ///
52    /// If pedantic is true, then an error will be returned as soon as anything
53    /// is missing in the file, or two values in the image contradict each
54    /// other. If pedantic is false, then only fatal errors will be thrown.
55    /// By default, reading an image is not pedantic, which means that
56    /// slightly invalid files might still be readable. For example, if some
57    /// attribute is missing but can be recomputed, this flag decides whether an
58    /// error is thrown. Or if the pedantic flag is true and there are still
59    /// bytes left after the decompression algorithm finished, an error is
60    /// thrown, because this should not happen and something might be wrong with
61    /// the file. Or if your application is a target of attacks, or if you
62    /// want to emulate the original C++ library, you might want to switch
63    /// to pedantic reading.
64    pub fn pedantic(self) -> Self {
65        Self {
66            pedantic: true,
67            ..self
68        }
69    }
70
71    /// Specify that multiple pixel blocks should never be decompressed using
72    /// multiple threads at once. This might be slower but uses less memory
73    /// and less synchronization.
74    pub fn non_parallel(self) -> Self {
75        Self {
76            parallel: false,
77            ..self
78        }
79    }
80
81    /// Specify a function to be called regularly throughout the loading
82    /// process. Replaces all previously specified progress functions in
83    /// this reader.
84    pub fn on_progress<OnProgress>(self, on_progress: OnProgress) -> ReadImage<OnProgress, L>
85    where
86        OnProgress: FnMut(f64),
87    {
88        ReadImage {
89            on_progress,
90            read_layers: self.read_layers,
91            pedantic: self.pedantic,
92            parallel: self.parallel,
93        }
94    }
95
96    /// Read the exr image from a file.
97    /// Use [`ReadImage::read_from_unbuffered`] instead, if you do not have a
98    /// file.
99    #[inline]
100    #[must_use]
101    pub fn from_file<Layers>(self, path: impl AsRef<Path>) -> Result<Image<Layers>>
102    where
103        for<'s> L: ReadLayers<'s, Layers = Layers>,
104    {
105        self.from_unbuffered(std::fs::File::open(path)?)
106    }
107
108    /// Buffer the reader and then read the exr image from it.
109    /// Use [`ReadImage::read_from_buffered`] instead, if your reader is an
110    /// in-memory reader. Use [`ReadImage::read_from_file`] instead, if you
111    /// have a file path.
112    #[inline]
113    #[must_use]
114    pub fn from_unbuffered<Layers>(self, unbuffered: impl Read + Seek) -> Result<Image<Layers>>
115    where
116        for<'s> L: ReadLayers<'s, Layers = Layers>,
117    {
118        self.from_buffered(BufReader::new(unbuffered))
119    }
120
121    /// Read the exr image from a buffered reader.
122    /// Use [`ReadImage::read_from_file`] instead, if you have a file path.
123    /// Use [`ReadImage::read_from_unbuffered`] instead, if this is not an
124    /// in-memory reader.
125    // TODO Use Parallel<> Wrapper to only require sendable byte source where parallel decompression
126    // is required
127    #[must_use]
128    pub fn from_buffered<Layers>(self, buffered: impl Read + Seek) -> Result<Image<Layers>>
129    where
130        for<'s> L: ReadLayers<'s, Layers = Layers>,
131    {
132        let chunks = crate::block::read(buffered, self.pedantic)?;
133        self.from_chunks(chunks)
134    }
135
136    /// Read the exr image from an initialized chunks reader
137    /// that has already extracted the meta data from the file.
138    /// Use [`ReadImage::read_from_file`] instead, if you have a file path.
139    /// Use [`ReadImage::read_from_buffered`] instead, if this is an in-memory
140    /// reader.
141    // TODO Use Parallel<> Wrapper to only require sendable byte source where parallel decompression
142    // is required
143    #[must_use]
144    pub fn from_chunks<Layers>(
145        mut self,
146        chunks_reader: crate::block::reader::Reader<impl Read + Seek>,
147    ) -> Result<Image<Layers>>
148    where
149        for<'s> L: ReadLayers<'s, Layers = Layers>,
150    {
151        let Self {
152            pedantic,
153            parallel,
154            ref mut on_progress,
155            ref mut read_layers,
156        } = self;
157
158        let layers_reader = read_layers.create_layers_reader(chunks_reader.headers())?;
159        let mut image_collector =
160            ImageWithAttributesReader::new(chunks_reader.headers(), layers_reader)?;
161
162        let block_reader = chunks_reader
163            .filter_chunks(pedantic, |meta, tile, block| {
164                image_collector.filter_block(meta, tile, block)
165            })?
166            .on_progress(on_progress);
167
168        // TODO propagate send requirement further upwards
169        if parallel {
170            #[cfg(not(feature = "rayon"))]
171            return Err(crate::error::Error::unsupported(
172                "parallel decompression requires the rayon feature",
173            ));
174
175            #[cfg(feature = "rayon")]
176            block_reader.decompress_parallel(pedantic, |meta_data, block| {
177                image_collector.read_block(&meta_data.headers, block)
178            })?;
179        } else {
180            block_reader.decompress_sequential(pedantic, |meta_data, block| {
181                image_collector.read_block(&meta_data.headers, block)
182            })?;
183        }
184
185        Ok(image_collector.into_image())
186    }
187}

189/// Processes blocks from a file and collects them into a complete `Image`.
190#[derive(Debug, Clone, PartialEq)]
191pub struct ImageWithAttributesReader<L> {
192    image_attributes: ImageAttributes,
193    layers_reader: L,
194}
195
196impl<L> ImageWithAttributesReader<L>
197where
198    L: LayersReader,
199{
200    /// A new image reader with image attributes.
201    pub fn new(headers: &[Header], layers_reader: L) -> Result<Self> {
202        Ok(ImageWithAttributesReader {
203            image_attributes: headers
204                .first()
205                .as_ref()
206                .expect("invalid headers")
207                .shared_attributes
208                .clone(),
209            layers_reader,
210        })
211    }
212
213    /// Specify whether a single block of pixels should be loaded from the file
214    fn filter_block(&self, meta: &MetaData, tile: TileCoordinates, block: BlockIndex) -> bool {
215        self.layers_reader.filter_block(meta, tile, block)
216    }
217
218    /// Load a single pixel block, which has not been filtered, into the reader,
219    /// accumulating the image
220    fn read_block(&mut self, headers: &[Header], block: UncompressedBlock) -> UnitResult {
221        self.layers_reader.read_block(headers, block)
222    }
223
224    /// Deliver the complete accumulated image
225    fn into_image(self) -> Image<L::Layers> {
226        Image {
227            attributes: self.image_attributes,
228            layer_data: self.layers_reader.into_layers(),
229        }
230    }
231}

233/// A template that creates a `LayerReader` for each layer in the file.
234pub trait ReadLayers<'s> {
235    /// The type of the resulting Layers
236    type Layers;
237
238    /// The type of the temporary layer reader
239    type Reader: LayersReader<Layers = Self::Layers>;
240
241    /// Create a single reader for a single layer
242    fn create_layers_reader(&'s self, headers: &[Header]) -> Result<Self::Reader>;
243
244    /// Specify that all attributes should be read from an image.
245    /// Use `from_file(path)` on the return value of this method to actually
246    /// decode an image.
247    fn all_attributes(self) -> ReadImage<fn(f64), Self>
248    where
249        Self: Sized,
250    {
251        ReadImage::new(self, ignore_progress)
252    }
253}

255/// Processes pixel blocks from a file and accumulates them into a single image
256/// layer.
257pub trait LayersReader {
258    /// The type of resulting layers
259    type Layers;
260
261    /// Specify whether a single block of pixels should be loaded from the file
262    fn filter_block(&self, meta: &MetaData, tile: TileCoordinates, block: BlockIndex) -> bool;
263
264    /// Load a single pixel block, which has not been filtered, into the reader,
265    /// accumulating the layer
266    fn read_block(&mut self, headers: &[Header], block: UncompressedBlock) -> UnitResult;
267
268    /// Deliver the final accumulated layers for the image
269    fn into_layers(self) -> Self::Layers;
270}
