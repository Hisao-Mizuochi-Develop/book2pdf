38    /// Read only layers that contain rgba channels. Skips any other channels in
39    /// the layer. The alpha channel will contain the value `1.0` if no
40    /// alpha channel can be found in the image.
41    ///
42    /// Using two closures, define how to store the pixels.
43    /// The first closure creates an image, and the second closure inserts a
44    /// single pixel. The type of the pixel can be defined by the second
45    /// closure; it must be a tuple containing four values, each being
46    /// either `f16`, `f32`, `u32` or `Sample`.
47    ///
48    /// Throws an error for images with deep data or subsampling.
49    /// Use `specific_channels` or `all_channels` if you want to read something
50    /// other than rgba.
51    pub fn rgba_channels<R, G, B, A, Create, Set, Pixels>(
52        self,
53        create_pixels: Create,
54        set_pixel: Set,
55    ) -> CollectPixels<
56        ReadOptionalChannel<
57            ReadRequiredChannel<ReadRequiredChannel<ReadRequiredChannel<NoneMore, R>, G>, B>,
58            A,
59        >,
60        (R, G, B, A),
61        Pixels,
62        Create,
63        Set,
64    >
65    where
66        R: FromNativeSample,
67        G: FromNativeSample,
68        B: FromNativeSample,
69        A: FromNativeSample,
70        Create: Fn(Vec2<usize>, &RgbaChannels) -> Pixels,
71        Set: Fn(&mut Pixels, Vec2<usize>, (R, G, B, A)),
72    {
73        self.specific_channels()
74            .required("R")
75            .required("G")
76            .required("B")
77            .optional("A", A::from_f32(1.0))
78            .collect_pixels(create_pixels, set_pixel)
79    }
80
81    /// Read only layers that contain rgb channels. Skips any other channels in
82    /// the layer.
83    ///
84    /// Using two closures, define how to store the pixels.
85    /// The first closure creates an image, and the second closure inserts a
86    /// single pixel. The type of the pixel can be defined by the second
87    /// closure; it must be a tuple containing three values, each being
88    /// either `f16`, `f32`, `u32` or `Sample`.
89    ///
90    /// Throws an error for images with deep data or subsampling.
91    /// Use `specific_channels` or `all_channels` if you want to read something
92    /// other than rgb.
93    pub fn rgb_channels<R, G, B, Create, Set, Pixels>(
94        self,
95        create_pixels: Create,
96        set_pixel: Set,
97    ) -> CollectPixels<
98        ReadRequiredChannel<ReadRequiredChannel<ReadRequiredChannel<NoneMore, R>, G>, B>,
99        (R, G, B),
100        Pixels,
101        Create,
102        Set,
103    >
104    where
105        R: FromNativeSample,
106        G: FromNativeSample,
107        B: FromNativeSample,
108        Create: Fn(Vec2<usize>, &RgbChannels) -> Pixels,
109        Set: Fn(&mut Pixels, Vec2<usize>, (R, G, B)),
110    {
111        self.specific_channels()
112            .required("R")
113            .required("G")
114            .required("B")
115            .collect_pixels(create_pixels, set_pixel)
116    }
117
118    /// Read only layers that contain the specified channels, skipping any other
119    /// channels in the layer. Further specify which channels should be
120    /// included by calling `.required("ChannelName")` or `.optional("
121    /// ChannelName", default_value)` on the result of this function.
122    /// Call `collect_pixels` afterwards to define the pixel container for your
123    /// set of channels.
124    ///
125    /// Throws an error for images with deep data or subsampling.
126    pub fn specific_channels(self) -> ReadZeroChannels {
127        ReadZeroChannels {}
128    }

153/// Processes pixel blocks from a file and accumulates them into multiple levels
154/// per channel.
155#[derive(Debug, Clone, Eq, PartialEq)]
156pub struct AllLevelsReader<SamplesReader> {
157    levels: Levels<SamplesReader>,
158}

160/// A template that creates a [`SamplesReader`] once for each resolution level.
161pub trait ReadSamplesLevel {
162    /// The type of the temporary level reader
163    type Reader: SamplesReader;
164
165    /// Create a single reader for a single resolution level
166    fn create_samples_level_reader(
167        &self,
168        header: &Header,
169        channel: &ChannelDescription,
170        level: Vec2<usize>,
171        resolution: Vec2<usize>,
172    ) -> Result<Self::Reader>;
173}
174
175impl<S: ReadSamplesLevel> ReadSamples for ReadAllLevels<S> {
176    type Reader = AllLevelsReader<S::Reader>;
177
178    fn create_sample_reader(
179        &self,
180        header: &Header,
181        channel: &ChannelDescription,
182    ) -> Result<Self::Reader> {
183        let data_size = header.layer_size / channel.sampling;
184
185        let levels = {
186            if let crate::meta::BlockDescription::Tiles(tiles) = &header.blocks {
187                match tiles.level_mode {
188                    LevelMode::Singular => {
189                        Levels::Singular(self.read_samples.create_samples_level_reader(
190                            header,
191                            channel,
192                            Vec2(0, 0),
193                            header.layer_size,
194                        )?)
195                    }
196
197                    LevelMode::MipMap => Levels::Mip {
198                        rounding_mode: tiles.rounding_mode,
199                        level_data: {
200                            let round = tiles.rounding_mode;
201                            let maps: Result<LevelMaps<S::Reader>> =
202                                mip_map_levels(round, data_size)
203                                    .map(|(index, level_size)| {
204                                        self.read_samples.create_samples_level_reader(
205                                            header,
206                                            channel,
207                                            Vec2(index, index),
208                                            level_size,
209                                        )
210                                    })
211                                    .collect();
212
213                            maps?
214                        },
215                    },
216
217                    // TODO put this into Levels::new(..) ?
218                    LevelMode::RipMap => Levels::Rip {
219                        rounding_mode: tiles.rounding_mode,
220                        level_data: {
221                            let round = tiles.rounding_mode;
222                            let level_count_x = compute_level_count(round, data_size.width());
223                            let level_count_y = compute_level_count(round, data_size.height());
224                            let maps: Result<LevelMaps<S::Reader>> =
225                                rip_map_levels(round, data_size)
226                                    .map(|(index, level_size)| {
227                                        self.read_samples.create_samples_level_reader(
228                                            header, channel, index, level_size,
229                                        )
230                                    })
231                                    .collect();
232
233                            RipMaps {
234                                map_data: maps?,
235                                level_count: Vec2(level_count_x, level_count_y),
236                            }
237                        },
238                    },
239                }
240            }
241            // scan line blocks never have mip maps
242            else {
243                Levels::Singular(self.read_samples.create_samples_level_reader(
244                    header,
245                    channel,
246                    Vec2(0, 0),
247                    data_size,
248                )?)
249            }
250        };
251
252        Ok(AllLevelsReader {
253            levels,
254        })
255    }
256}

258impl<S: SamplesReader> SamplesReader for AllLevelsReader<S> {
259    type Samples = Levels<S::Samples>;
260
261    fn filter_block(&self, _: TileCoordinates) -> bool {
262        true
263    }
264
265    fn read_line(&mut self, line: LineRef<'_>) -> UnitResult {
266        self.levels.get_level_mut(line.location.level)?.read_line(line)
267    }
268
269    fn into_samples(self) -> Self::Samples {
270        match self.levels {
271            Levels::Singular(level) => Levels::Singular(level.into_samples()),
272            Levels::Mip {
273                rounding_mode,
274                level_data,
275            } => Levels::Mip {
276                rounding_mode,
277                level_data: level_data.into_iter().map(|s| s.into_samples()).collect(),
278            },
279
280            Levels::Rip {
281                rounding_mode,
282                level_data,
283            } => Levels::Rip {
284                rounding_mode,
285                level_data: RipMaps {
286                    level_count: level_data.level_count,
287                    map_data: level_data.map_data.into_iter().map(|s| s.into_samples()).collect(),
288                },
289            },
290        }
291    }
