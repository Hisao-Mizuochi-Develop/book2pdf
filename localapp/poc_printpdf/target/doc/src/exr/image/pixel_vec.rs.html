1//! Provides a predefined pixel storage.
2//! Currently only contains a simple flattened vector storage.
3//! Use the functions `create_pixel_vec::<YourPixelTuple>` and
4//! `set_pixel_in_vec::<YourPixelTuple>` for reading a predefined pixel vector.
5//! Use the function `PixelVec::new` to create a pixel vector which can be
6//! written to a file.
7
8use super::*;

10/// Store all samples in a single array.
11/// All samples will be converted to the type `T`.
12/// This supports all the sample types, `f16`, `f32`, and `u32`.
13///
14/// The flattened vector contains all rows one after another.
15/// In each row, for each pixel, its red, green, blue, and then alpha
16/// samples are stored one after another.
17///
18/// Use `PixelVec.compute_pixel_index(position)`
19/// to compute the flat index of a specific pixel.
20#[derive(Eq, PartialEq, Clone)]
21pub struct PixelVec<T> {
22    /// The resolution of this layer.
23    pub resolution: Vec2<usize>,
24
25    /// The flattened vector contains all rows one after another.
26    /// In each row, for each pixel, its red, green, blue, and then alpha
27    /// samples are stored one after another.
28    ///
29    /// Use `Flattened::compute_pixel_index(image, position)`
30    /// to compute the flat index of a specific pixel.
31    pub pixels: Vec<T>,
32}
33
34impl<Pixel> PixelVec<Pixel> {
35    /// Create a new flattened pixel storage, filled with default pixels.
36    /// Accepts a `Channels` parameter, which is not used, so that it can be
37    /// passed as a function pointer instead of calling it.
38    #[must_use]
39    pub fn constructor<Channels>(resolution: Vec2<usize>, _: &Channels) -> Self
40    where
41        Pixel: Default + Clone,
42    {
43        Self {
44            resolution,
45            pixels: vec![Pixel::default(); resolution.area()],
46        }
47    }
48
49    /// Examine a pixel of a `PixelVec<T>` image.
50    /// Can usually be used as a function reference instead of calling it
51    /// directly.
52    #[inline]
53    pub fn get_pixel(&self, position: Vec2<usize>) -> &Pixel
54    where
55        Pixel: Sync,
56    {
57        &self.pixels[self.compute_pixel_index(position)]
58    }
59
60    /// Update a pixel of a `PixelVec<T>` image.
61    /// Can usually be used as a function reference instead of calling it
62    /// directly.
63    #[inline]
64    pub fn set_pixel(&mut self, position: Vec2<usize>, pixel: Pixel) {
65        let index = self.compute_pixel_index(position);
66        self.pixels[index] = pixel;
67    }
68
69    /// Create a new flattened pixel storage, checking the length of the
70    /// provided pixels vector.
71    pub fn new(resolution: impl Into<Vec2<usize>>, pixels: Vec<Pixel>) -> Self {
72        let size = resolution.into();
73        assert_eq!(
74            size.area(),
75            pixels.len(),
76            "expected {} samples, but vector length is {}",
77            size.area(),
78            pixels.len()
79        );
80        Self {
81            resolution: size,
82            pixels,
83        }
84    }
85
86    /// Compute the flat index of a specific pixel. Returns a range of either 3
87    /// or 4 samples. The computed index can be used with
88    /// `PixelVec.samples[index]`. Panics for invalid sample coordinates.
89    #[inline]
90    pub fn compute_pixel_index(&self, position: Vec2<usize>) -> usize {
91        position.flat_index_for_size(self.resolution)
92    }
93}

95use crate::image::validate_results::{ValidateResult, ValidationResult};
96
97impl<Px> ValidateResult for PixelVec<Px>
98where
99    Px: ValidateResult,
100{
101    fn validate_result(
102        &self,
103        other: &Self,
104        options: ValidationOptions,
105        location: impl Fn() -> String,
106    ) -> ValidationResult {
107        if self.resolution == other.resolution {
108            self.pixels
109                .as_slice()
110                .validate_result(&other.pixels.as_slice(), options, || location() + " > pixels")
111        } else {
112            Err(location() + " > resolution")
113        }
114    }
115}
